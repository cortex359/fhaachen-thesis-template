% Bibliography field configuration
\DeclareSourcemap{
	\maps[datatype=bibtex, overwrite]{
    % map paywalled attribute to userb for sci-hub linking
    \map{
      \step[fieldsource=paywalled, final]
      \step[fieldset=userb, origfieldval]
    }
		% always remove unused fields
		\map{
			\step[fieldset=abstract, null]
			\step[fieldset=editor, null]  % keep only author, journal, publisher
			\step[fieldset=isbn, null]
			\step[fieldset=month, null]   % year is sufficient
			\step[fieldset=number, null]  % volume is sufficient
			\step[fieldset=pagetotal, null]
		}
		% if DOI exists, drop URL and eprint
		\map{
			\step[fieldsource=doi, final]
			\step[fieldset=eprint, null]
			\step[fieldset=url, null]
			\step[fieldset=urldate, null]
		}
		% if no DOI but eprint exists, also drop URL
		\map{
			\step[notfield=doi, final]
			\step[fieldsource=eprint, final]
			\step[fieldset=url, null]
			\step[fieldset=urldate, null]
		} % if neither DOI nor eprint exists, URL is kept
		% remove redundant fields for conference papers
		\map{
			\pertype{inproceedings}
			\step[fieldset=venue, null]
			\step[fieldset=eventdate, null]
			\step[fieldset=eventtitle, null]
		}
		% remove redundant fields for books
		\map{
			\pertype{book}
			\step[fieldset=url, null]
		}
	}
}
